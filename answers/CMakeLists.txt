cmake_minimum_required(VERSION 3.13)
# Default: four host reference programs. Select one Pico answer with
# -DANSWER_PICO=1..6, using a separate build directory.
set(ANSWER_PICO "0" CACHE STRING "0 for host answers; 1..6 for a Pico answer")
set(LABS "${CMAKE_CURRENT_LIST_DIR}/..")
if (ANSWER_PICO STREQUAL "0")
    project(bughunt_answers C)
    enable_testing()
    add_executable(bughunt1_answer bughunt1/bughunt1.c)
    add_executable(bughunt2_answer bughunt2/frame.c "${LABS}/LAB2/bughunt/bughunt2_host.c")
    target_include_directories(bughunt2_answer PRIVATE "${LABS}/LAB2/bughunt")
    add_executable(bughunt4_answer bughunt4/algo.c "${LABS}/LAB4/bughunt/bughunt4_host.c")
    target_include_directories(bughunt4_answer PRIVATE "${LABS}/LAB4/bughunt")
    add_executable(bughunt6_answer bughunt6/bughunt6.c "${LABS}/LAB6/bughunt/bughunt6_host.c")
    foreach(n 1 2 4 6)
        target_compile_options(bughunt${n}_answer PRIVATE -Wall -Wextra)
        add_test(NAME bughunt${n} COMMAND bughunt${n}_answer)
    endforeach()
else()
    if (NOT ANSWER_PICO MATCHES "^[1-6]$")
        message(FATAL_ERROR "ANSWER_PICO must be 0 or 1..6")
    endif()
    if (NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
        set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
    endif()
    if (NOT PICO_SDK_PATH)
        message(FATAL_ERROR "Set PICO_SDK_PATH")
    endif()
    include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
    if (ANSWER_PICO STREQUAL "5")
        if (NOT FREERTOS_KERNEL_PATH AND DEFINED ENV{FREERTOS_KERNEL_PATH})
            set(FREERTOS_KERNEL_PATH "$ENV{FREERTOS_KERNEL_PATH}")
        endif()
        if (NOT FREERTOS_KERNEL_PATH)
            message(FATAL_ERROR "Set FREERTOS_KERNEL_PATH")
        endif()
        include("${FREERTOS_KERNEL_PATH}/portable/ThirdParty/GCC/RP2040/FreeRTOS_Kernel_import.cmake")
    endif()
    project(bughunt_answer C CXX ASM)
    pico_sdk_init()
    set(n "${ANSWER_PICO}")
    set(target "bughunt${n}_answer")
    if (n STREQUAL "2")
        add_executable(${target} bughunt2/bughunt2_pico.c bughunt2/frame.c)
        target_link_libraries(${target} hardware_uart)
    elseif(n STREQUAL "4")
        add_executable(${target} bughunt4/bughunt4.c bughunt4/algo.c)
        target_link_libraries(${target} hardware_pwm hardware_adc)
    elseif(n STREQUAL "6")
        add_executable(${target} bughunt6/bughunt6.c "${LABS}/LAB6/bughunt/bughunt6_pico.c")
    else()
        add_executable(${target} "bughunt${n}/bughunt${n}.c")
    endif()
    target_include_directories(${target} PRIVATE
        "${CMAKE_CURRENT_LIST_DIR}/bughunt${n}" "${LABS}/LAB${n}/bughunt")
    if (n STREQUAL "5")
        target_link_libraries(${target} hardware_adc FreeRTOS-Kernel FreeRTOS-Kernel-Heap4)
    endif()
    target_link_libraries(${target} pico_stdlib)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wno-unused-parameter)
    pico_enable_stdio_usb(${target} 1)
    pico_enable_stdio_uart(${target} 0)
    pico_add_extra_outputs(${target})
endif()
